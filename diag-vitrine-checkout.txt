23: export function useUpsells(modules: string[]) {
24:   const [items, setItems] = useState<Upsell[]>([]);
25:   const key = modules.join('|');
27:   useEffect(() => {
28:     let alive = true;
29:     sb.from('addons')
30:       .select('id,name,description,price_monthly,scope,upsell_headline,upsell_module_target')
31:       .eq('is_active', true)
32:       .eq('is_upsell', true)
33:       .order('price_monthly')
34:       .then(({ data }: any) => {
35:         if (!alive) return;
36:         const mods = key ? key.split('|') : [];
37:         setItems(
38:           ((data || []) as any[])
39:             .map((r) => ({ ...r, price_monthly: Number(r.price_monthly) }) as Upsell)
40:             .filter((a) => (a.scope === 'global' || mods.includes(a.scope)) && (a.upsell_module_target === 'all' || (a.upsell_module_target && mods.includes(a.upsell_module_target)))),
41:         );
42:       });
43:     return () => { alive = false; };
44:   }, [key]);
46:   return items;
47: }
