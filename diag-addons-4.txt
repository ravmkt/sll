211: async function syncInvoice(eventType: string, payment: any, sub: Sub | null) {
212:   const target = INVOICE_STATUS_MAP[eventType];
213:   const paymentId: string | undefined = payment?.id;
214:   if (!target || !paymentId) return;
215: 
216:   const { data: inv, error: invErr } = await supabase
217:     .from("invoices")
218:     .select("id, status, payment_method, invoice_url")
219:     .eq("asaas_payment_id", paymentId)
220:     .maybeSingle();
221:   if (invErr) {
222:     console.error("Erro ao buscar invoice:", invErr);
223:     return;
224:   }
225: 
226:   const now = new Date().toISOString();
227: 
228:   if (inv) {
229:     // meio real so aparece apos o cliente escolher a forma de pagamento (antes vem UNDEFINED)
230:     const bt = payment?.billingType as string | undefined;
231:     const meta: Record<string, unknown> = {};
232:     if (bt && bt !== "UNDEFINED" && inv.payment_method !== bt) meta.payment_method = bt;
233:     if (payment?.invoiceUrl && !inv.invoice_url) meta.invoice_url = payment.invoiceUrl;
234:     if (Object.keys(meta).length > 0) {
235:       const { error: metaErr } = await supabase.from("invoices").update({ ...meta, updated_at: now }).eq("id", inv.id);
236:       if (metaErr) console.error("Erro ao atualizar meio/link da invoice:", metaErr);
237:     }
238:     if (inv.status === target || inv.status === "refunded") return;
239:     // evento atrasado/fora de ordem nao desfaz uma fatura ja paga
240:     if (inv.status === "paid" && target !== "refunded") return;
241:     const patch: Record<string, unknown> = { status: target, updated_at: now };
