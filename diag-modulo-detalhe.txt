1: import { useCallback, useEffect, useMemo, useRef, useState } from 'react';
2: import { ArrowLeft, Loader2 } from 'lucide-react';
3: import { toast } from 'sonner';
4: import ModulePlansCard from '@/components/admin/ModulePlansCard';
5: import FeatureFlagsCard from '@/components/admin/FeatureFlagsCard';
6: import ModuloVitrineEditor, { parseFeatures } from '@/components/admin/ModuloVitrineEditor';
7: import StagingActionBar from '@/components/master/StagingActionBar';
8: import { useStagingPublish } from '@/hooks/useStagingPublish';
9: import {
10:   getModuleStores, getModulesOverview, updateHubModule, uploadModuleLogo,
11:   type ModuleOverview, type ModuleStoreRow,
12: } from '@/services/admin/modulesService';
13: import {
14:   CARD, CYCLE_LABEL, INPUT, SELECT, SUB_LABEL, Kpi, ModuleLogo, bytes, brl, int,
15: } from '@/components/admin/moduleUi';
16: 
17: const TH = 'px-3 py-2 text-left text-[10px] font-bold uppercase tracking-wider text-slate-500';
18: 
19: type VitrineCfg = {
20:   name: string; status: string; is_public_for_sale: boolean;
21:   description: string; features: string[]; logo_url: string | null;
22: };
23: 
24: export default function ModuloDetalhe({ slug, onBack, onOpenStore, onOpenPlans }: {
25:   slug: string; onBack: () => void; onOpenStore: (id: string) => void; onOpenPlans: (slug: string) => void;
26: }) {
27:   const [days, setDays] = useState(30);
28:   const [m, setM] = useState<ModuleOverview | null>(null);
29:   const [stores, setStores] = useState<ModuleStoreRow[] | null>(null);
30:   const [loading, setLoading] = useState(true);
31:   const [error, setError] = useState('');
32:   const [q, setQ] = useState('');
33: 
34:   const [name, setName] = useState('');
35:   const [status, setStatus] = useState('active');
36:   const [pub, setPub] = useState(false);
37:   const [sort, setSort] = useState('1');
38:   const [logo, setLogo] = useState<string | null>(null);
39:   const [description, setDescription] = useState('');
40:   const [featuresText, setFeaturesText] = useState('');
41:   const [saving, setSaving] = useState(false);
42:   const [uploading, setUploading] = useState(false);
43:   const filled = useRef(false);
44:   const fileRef = useRef<HTMLInputElement>(null);
45: 
46:   const load = useCallback(async (fill: boolean) => {
47:     setLoading(true);
48:     setError('');
49:     try {
50:       const res = await getModulesOverview(days);
51:       const found = res.modules.find((x) => x.slug === slug) || null;
52:       if (!found) { setError('Módulo não encontrado.'); return; }
53:       setM(found);
54:       if (fill || !filled.current) {
55:         setName(found.name);
56:         setStatus(found.status);
57:         setPub(found.is_public_for_sale);
58:         setSort(String(found.sort_order ?? 1));
59:         setLogo(found.logo_url);
60:         setDescription(found.description ?? '');
61:         setFeaturesText((found.features ?? []).join('\n'));
62:         filled.current = true;
63:       }
64:     } catch (e: any) {
65:       setError(e?.message || 'Erro ao carregar o módulo.');
66:     } finally {
67:       setLoading(false);
68:     }
69:   }, [days, slug]);
70: 
71:   useEffect(() => { load(false); }, [load]);
72: 
73:   useEffect(() => {
74:     getModuleStores(slug).then(setStores).catch((e: any) => { toast.error(e?.message || 'Erro ao listar lojas.'); setStores([]); });
75:   }, [slug]);
76: 
77:   const onFile = async (file: File | undefined) => {
78:     if (!file) return;
79:     setUploading(true);
80:     try { setLogo(await uploadModuleLogo(slug, file)); toast.success('Logo enviado. Clique em Aplicar na Loja Teste.'); }
81:     catch (e: any) { toast.error(e?.message || 'Falha no envio do logo.'); }
82:     finally { setUploading(false); if (fileRef.current) fileRef.current.value = ''; }
83:   };
84: 
85:   const save = async () => {
86:     if (!m) return;
87:     setSaving(true);
88:     try {
89:       const logoArg = logo === (m.logo_url ?? null) ? null : (logo ?? '');
90:       const feats = parseFeatures(featuresText);
91:       await updateHubModule(slug, name, status, status === 'active' ? pub : false, Number(sort) || 1, logoArg, description, feats);
92:       toast.success('Módulo atualizado.');
93:       await load(true);
94:     } catch (e: any) {
95:       toast.error(e?.message || 'Não foi possível salvar.');
96:     } finally {
97:       setSaving(false);
98:     }
99:   };
100: 
101:   const liveCfg = useMemo<VitrineCfg | undefined>(() => (m ? {
102:     name: m.name, status: m.status, is_public_for_sale: m.is_public_for_sale,
103:     description: m.description ?? '', features: m.features ?? [], logo_url: m.logo_url ?? null,
104:   } : undefined), [m]);
105: 
106:   const formCfg: VitrineCfg = {
107:     name: name.trim(), status, is_public_for_sale: status === 'active' ? pub : false,
108:     description, features: parseFeatures(featuresText), logo_url: logo,
109:   };
110: 
111:   const staging = useStagingPublish<VitrineCfg>('module', slug, liveCfg, {
112:     applyLive: async (c) => {
113:       const logoArg = c.logo_url === (m?.logo_url ?? null) ? null : (c.logo_url ?? '');
114:       await updateHubModule(slug, c.name, c.status, c.status === 'active' ? c.is_public_for_sale : false, Number(m?.sort_order) || 1, logoArg, c.description, c.features);
115:     },
116:     onSettled: (k) => { if (k !== 'save') void load(true); },
117:   });
118: 
119:   const draftApplied = useRef(false);
120:   useEffect(() => {
121:     if (!m || staging.loading || draftApplied.current) return;
122:     draftApplied.current = true;
123:     const d = staging.hasDraft ? staging.test : null;
124:     if (!d) return;
125:     setName(d.name); setStatus(d.status); setPub(!!d.is_public_for_sale);
126:     setDescription(d.description ?? ''); setFeaturesText((d.features ?? []).join('\n')); setLogo(d.logo_url ?? null);
127:   }, [m, staging.loading, staging.hasDraft, staging.test]);
128: 
129:   const term = q.trim().toLowerCase();
130:   const shown = (stores || []).filter((x) => !term || (x.name || '').toLowerCase().includes(term));
131:   const measured = !!m && m.events !== null;
132:   const ctr = m && m.views ? ((Number(m.clicks) || 0) / m.views) * 100 : 0;
133: 
134:   return (
135:     <div className="space-y-5">
136:       <div className="sticky top-0 z-20 -mx-6 -mt-6 flex flex-wrap items-center justify-between gap-3 border-b border-slate-800 bg-[#0b0e1a] px-6 pb-3 pt-6">
137:         <div className="flex items-center gap-3">
138:           <button type="button" onClick={onBack} className="flex cursor-pointer items-center gap-1.5 rounded-lg bg-white/5 px-3 py-1.5 text-xs font-bold text-slate-200 hover:bg-white/10">
139:             <ArrowLeft size={14} /> Módulos
140:           </button>
141:           {m && <ModuleLogo url={logo} name={name || m.name} size={40} />}
142:           <div>
143:             <h1 className="text-xl font-black text-white">{m?.name || 'Módulo'}</h1>
144:             <p className="font-mono text-[10px] text-slate-600">{slug}</p>
145:           </div>
146:         </div>
147:         <div className="flex items-center gap-2">
148:           {loading && <Loader2 className="h-4 w-4 animate-spin text-[#fd8539]" />}
149:           <select value={days} onChange={(e) => setDays(Number(e.target.value))} className={`${SELECT} cursor-pointer`}>
150:             <option value={7}>Últimos 7 dias</option>
151:             <option value={30}>Últimos 30 dias</option>
152:             <option value={90}>Últimos 90 dias</option>
153:           </select>
154:         </div>
155:       </div>
156: 
157:       {error && <div className="rounded-xl border border-rose-500/30 bg-rose-500/10 px-4 py-3 text-sm text-rose-300">{error}</div>}
158: 
159:       {m && (
160:         <>
161:         <ModuloVitrineEditor
162:           slug={slug}
163:           name={name} setName={setName}
164:           status={status} setStatus={setStatus}
165:           pub={pub} setPub={setPub}
166:           sort={sort} setSort={setSort}
167:           logo={logo} setLogo={setLogo}
168:           description={description} setDescription={setDescription}
169:           featuresText={featuresText} setFeaturesText={setFeaturesText}
170:           fileRef={fileRef} onFile={onFile}
171:           uploading={uploading} saving={saving} onSave={save}
172:           footer={<StagingActionBar staging={staging} config={formCfg} disabled={uploading || !name.trim()} />}
173:         />
174:         <div className="grid grid-cols-1 gap-4 xl:grid-cols-3">
175:           <div className="space-y-4 xl:col-span-2">
176:             <div className={`${CARD} space-y-3`}>
177:               <p className="text-sm font-bold text-white">Lojas e receita</p>
178:               <div className="grid grid-cols-2 gap-2 md:grid-cols-3">
179:                 <Kpi label="Com acesso" value={int(m.access)} />
180:                 <Kpi label="Pagantes" value={int(m.paying)} tone="text-emerald-300" />
181:                 <Kpi label="Em trial" value={int(m.trial)} tone="text-sky-300" />
182:                 <Kpi label="Inadimplentes" value={int(m.past_due)} tone={m.past_due ? 'text-rose-300' : undefined} />
183:                 <Kpi label="Vitalícias" value={int(m.lifetime)} />
184:                 <Kpi label="Receita mensal" value={brl(m.mrr)} />
185:               </div>
186:             </div>
187: 
188:             <div className={`${CARD} space-y-3`}>
189:               <p className="text-sm font-bold text-white">Consumo e desempenho ({days} dias)</p>
190:               {measured ? (
191:                 <div className="grid grid-cols-2 gap-2 md:grid-cols-3">
192:                   <Kpi label="Views" value={int(m.views)} />
193:                   <Kpi label="Cliques" value={int(m.clicks)} />
194:                   <Kpi label="CTR" value={`${ctr.toFixed(1).replace('.', ',')}%`} />
195:                   <Kpi label="Lojas ativas" value={int(m.active_stores)} />
196:                   <Kpi label="Vídeos" value={int(m.videos)} />
197:                   <Kpi label="Espaço" value={bytes(m.storage_bytes)} />
198:                 </div>
199:               ) : (
200:                 <p className="rounded-xl bg-white/[0.03] px-3 py-3 text-xs text-slate-500">O consumo deste módulo ainda não é medido.</p>
201:               )}
202:               <p className={`text-[11px] ${m.errors ? 'text-rose-300' : 'text-slate-500'}`}>
203:                 {m.errors ? `${int(m.errors)} erro(s) registrado(s) no período.` : 'Nenhum erro registrado no período.'}
204:               </p>
205:             </div>
206: 
207:             <div className={`${CARD} space-y-3`}>
208:               <div className="flex flex-wrap items-center justify-between gap-2">
209:                 <p className="text-sm font-bold text-white">Lojas com acesso{stores ? ` (${stores.length})` : ''}</p>
210:                 <input value={q} onChange={(e) => setQ(e.target.value)} placeholder="Filtrar loja..." className={`${INPUT} max-w-[220px]`} />
211:               </div>
212:               {stores === null ? (
213:                 <Loader2 className="h-4 w-4 animate-spin text-[#fd8539]" />
214:               ) : stores.length === 0 ? (
215:                 <p className="text-xs text-slate-500">Nenhuma loja com acesso.</p>
216:               ) : (
217:                 <div className="overflow-x-auto rounded-xl border border-slate-800">
218:                   <table className="w-full min-w-[560px] text-xs">
219:                     <thead className="border-b border-slate-800">
220:                       <tr><th className={TH}>Loja</th><th className={TH}>Plano</th><th className={TH}>Ciclo</th><th className={TH}>Origem</th><th className={TH}>Status</th></tr>
221:                     </thead>
222:                     <tbody className="divide-y divide-white/5">
223:                       {shown.length === 0 && <tr><td colSpan={5} className="px-3 py-6 text-center text-slate-500">Nenhuma loja encontrada.</td></tr>}
224:                       {shown.map((s) => (
225:                         <tr key={s.id} onClick={() => onOpenStore(s.id)} className="cursor-pointer hover:bg-white/5">
226:                           <td className="px-3 py-2">
227:                             <button type="button" onClick={(e) => { e.stopPropagation(); onOpenStore(s.id); }} className="cursor-pointer text-left font-bold text-white underline-offset-2 hover:text-[#fd8539] hover:underline">
228:                               {s.name || 'Sem nome'}
229:                             </button>
230:                           </td>
231:                           <td className="px-3 py-2 text-slate-300">{s.plan_name || '—'}</td>
232:                           <td className="px-3 py-2 text-slate-400">{s.billing_cycle ? CYCLE_LABEL[s.billing_cycle] || s.billing_cycle : '—'}</td>
233:                           <td className="px-3 py-2 text-slate-400">{s.via_combo ? 'Combo' : 'Individual'}</td>
234:                           <td className="px-3 py-2 text-slate-300">{SUB_LABEL[s.status] || s.status}</td>
235:                         </tr>
236:                       ))}
237:                     </tbody>
238:                   </table>
239:                 </div>
240:               )}
241:             </div>
242:           </div>
243: 
244:           <div className="space-y-4">
245:             <ModulePlansCard slug={slug} onManage={() => onOpenPlans(slug)} />
246:             <FeatureFlagsCard slug={slug} />
247:           </div>
248:         </div>
249:         </>
250:       )}
251:     </div>
252:   );
253: }
