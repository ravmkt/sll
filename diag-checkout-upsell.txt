37: function validCpf(v: string) {
38:   const d = digits(v);
39:   if (d.length !== 11 || /^(\d)\1+$/.test(d)) return false;
40:   for (const t of [9, 10]) {
41:     let s = 0;
42:     for (let i = 0; i < t; i++) s += Number(d[i]) * (t + 1 - i);
43:     if (((s * 10) % 11) % 10 !== Number(d[t])) return false;
44:   }
45:   return true;
46: }
47: function validCnpj(v: string) {
48:   const d = digits(v);
49:   if (d.length !== 14 || /^(\d)\1+$/.test(d)) return false;
50:   const calc = (n: number) => {
51:     let s = 0, p = n - 7;
52:     for (let i = 0; i < n; i++) { s += Number(d[i]) * p--; if (p < 2) p = 9; }
53:     const r = s % 11;
54:     return r < 2 ? 0 : 11 - r;
55:   };
56:   return calc(12) === Number(d[12]) && calc(13) === Number(d[13]);
57: }
