1: import { useCallback, useEffect, useMemo, useState } from 'react';
2: import { Loader2, Pencil, Plus, Trash2, X } from 'lucide-react';
3: import { toast } from 'sonner';
4: import { supabase } from '@/lib/supabase';
5: import { useConfirm } from '@/components/admin/ConfirmDialog';
6: import {
7:   RESOURCE_LABEL, brl, deleteAddon, formatGrant, listAddons, saveAddon,
8:   type Addon, type AddonInput, type Resource,
9: } from '@/services/admin/addonsAdmin';
10: 
11: const sb: any = supabase;
12: 
13: const IN = 'w-full rounded-lg border border-slate-700 bg-[#0b0e1a] px-3 py-2 text-[15px] text-slate-100 outline-none focus:border-[#0094eb]';
14: const LBL = 'mb-1 block text-[13px] font-semibold uppercase tracking-wider text-slate-400';
15: const TH = 'px-3 py-2 text-xs font-bold uppercase tracking-wider text-slate-500';
16: 
17: type Mod = { slug: string; name: string };
18: 
19: type Form = {
20:   name: string; description: string; scope: string; target_resource: Resource; amount: string; price: string;
21:   is_active: boolean; is_upsell: boolean; upsell_module_target: string; upsell_headline: string;
22: };
23: 
24: const EMPTY: Form = {
25:   name: '', description: '', scope: 'global', target_resource: 'storage_gb', amount: '', price: '',
26:   is_active: true, is_upsell: false, upsell_module_target: 'all', upsell_headline: '',
27: };
28: 
29: const num = (v: string) => Number(v.trim().replace(/\./g, '').replace(',', '.'));
30: 
31: function Switch({ on, onChange, label }: { on: boolean; onChange: (v: boolean) => void; label: string }) {
32:   return (
33:     <label className="flex cursor-pointer items-center gap-3 text-[15px] text-slate-200">
34:       <button type="button" role="switch" aria-checked={on} onClick={() => onChange(!on)}
35:         className={`relative h-5 w-9 shrink-0 rounded-full transition-colors ${on ? 'bg-emerald-500' : 'bg-slate-600'}`}>
36:         <span className={`absolute top-0.5 h-4 w-4 rounded-full bg-white transition-all ${on ? 'left-[18px]' : 'left-0.5'}`} />
37:       </button>
38:       {label}
39:     </label>
40:   );
41: }

43: export default function AddonsAdminTab() {
44:   const [list, setList] = useState<Addon[]>([]);
45:   const [mods, setMods] = useState<Mod[]>([]);
46:   const [loading, setLoading] = useState(true);
47:   const [editId, setEditId] = useState<string | null>(null);
48:   const [form, setForm] = useState<Form | null>(null);
49:   const [busy, setBusy] = useState(false);
50:   const { confirm, dialog } = useConfirm();
51: 
52:   const load = useCallback(async () => {
53:     try { setList(await listAddons()); } catch (e) { toast.error((e as Error).message); } finally { setLoading(false); }
54:   }, []);
55: 
56:   useEffect(() => { load(); }, [load]);
57:   useEffect(() => {
58:     sb.from('hub_modules').select('slug,name').order('sort_order').then(({ data }: any) => setMods((data || []) as Mod[]));
59:   }, []);
60: 
61:   const scopes = useMemo(() => [{ slug: 'global', name: 'Global (Ecossistema SLL)' }, ...mods], [mods]);
62:   const targets = useMemo(
63:     () => [{ slug: 'all', name: 'Em todos os checkouts' }, ...mods.map((m) => ({ slug: m.slug, name: `Apenas ${m.name}` }))],
64:     [mods],
65:   );
66:   const scopeName = (s: string) => (s === 'global' ? 'Global' : mods.find((m) => m.slug === s)?.name ?? s);
67: 
68:   const set = <K extends keyof Form>(k: K, v: Form[K]) => setForm((f) => (f ? { ...f, [k]: v } : f));
69: 
70:   const openNew = () => { setEditId(null); setForm({ ...EMPTY }); };
71:   const openEdit = (a: Addon) => {
72:     setEditId(a.id);
73:     setForm({
74:       name: a.name, description: a.description ?? '', scope: a.scope, target_resource: a.target_resource,
75:       amount: String(a.resource_amount).replace('.', ','), price: String(a.price_monthly).replace('.', ','),
76:       is_active: a.is_active, is_upsell: a.is_upsell,
77:       upsell_module_target: a.upsell_module_target ?? 'all', upsell_headline: a.upsell_headline ?? '',
78:     });
79:   };
80: 
81:   const save = async () => {
82:     if (!form) return;
83:     const price = num(form.price || '0');
84:     const amount = form.target_resource === 'custom' ? num(form.amount || '0') : num(form.amount);
85:     if (form.name.trim().length < 3) return void toast.error('Informe o nome do add-on.');
86:     if (!Number.isFinite(price) || price < 0) return void toast.error('Preço mensal inválido.');
87:     if (!Number.isFinite(amount) || amount < 0 || (form.target_resource !== 'custom' && amount <= 0)) {
88:       return void toast.error('Informe a quantidade do recurso concedido.');
89:     }
90:     if (form.is_upsell) {
91:       if (form.upsell_headline.trim().length < 5) return void toast.error('Escreva a frase de destaque do upsell.');
92:       if (price <= 0) return void toast.error('Um upsell precisa ter preço maior que zero.');
93:     }
94:     const input: AddonInput = {
95:       name: form.name.trim(),
96:       description: form.description.trim() || null,
97:       price_monthly: price,
98:       scope: form.scope,
99:       target_resource: form.target_resource,
100:       resource_amount: amount,
101:       is_upsell: form.is_upsell,
102:       upsell_headline: form.is_upsell ? form.upsell_headline.trim() : null,
103:       upsell_module_target: form.is_upsell ? form.upsell_module_target : null,
104:       is_active: form.is_active,
105:     };
106:     setBusy(true);
107:     try {
108:       await saveAddon(input, editId ?? undefined);
109:       toast.success('Add-on salvo.');
110:       setForm(null);
111:       await load();
112:     } catch (e) {
113:       toast.error((e as Error).message);
114:     } finally {
115:       setBusy(false);
116:     }
117:   };
118: 
119:   const remove = async (a: Addon) => {
120:     if (!(await confirm({ title: 'Excluir add-on', message: `"${a.name}" será apagado.`, confirmLabel: 'Excluir' }))) return;
121:     try { await deleteAddon(a.id); toast.success('Add-on excluído.'); await load(); } catch (e) { toast.error((e as Error).message); }
122:   };
123: 
124:   return (
125:     <div className="space-y-4">
126:       <div className="flex flex-wrap items-center justify-between gap-3">
127:         <p className="text-[13px] text-slate-500">Pacotes extras de recurso. Os marcados como upsell aparecem como Order Bump no checkout.</p>
128:         <button type="button" onClick={openNew} className="flex cursor-pointer items-center gap-1.5 rounded-lg bg-[#fd8539] px-4 py-2 text-[13px] font-bold text-white">
129:           <Plus size={14} /> Novo Add-on
130:         </button>
131:       </div>
132: 
133:       <div className="overflow-x-auto rounded-2xl border border-slate-800 bg-[#111524]">
134:         <table className="w-full text-left text-[13px]">
135:           <thead className="border-b border-slate-800">
136:             <tr>
137:               <th className={TH}>Escopo / Chave</th><th className={TH}>Nome</th><th className={TH}>Recurso concedido</th>
138:               <th className={`${TH} text-center`}>Mensal (R$)</th><th className={`${TH} text-center`}>Upsell</th>
139:               <th className={`${TH} text-center`}>Status</th><th className={`${TH} text-center`}>Ações</th>
140:             </tr>
141:           </thead>
142:           <tbody className="divide-y divide-slate-800">
143:             {loading && <tr><td colSpan={7} className="px-3 py-8 text-center"><Loader2 className="mx-auto h-4 w-4 animate-spin text-[#fd8539]" /></td></tr>}
144:             {!loading && list.length === 0 && <tr><td colSpan={7} className="px-3 py-8 text-center text-slate-500">Nenhum add-on cadastrado.</td></tr>}
145:             {list.map((a) => (
146:               <tr key={a.id} className={a.is_active ? '' : 'opacity-50'}>
147:                 <td className="px-3 py-3">
148:                   <span className={`rounded-full px-2 py-0.5 text-xs font-bold ${a.scope === 'global' ? 'bg-violet-500/15 text-violet-300' : 'bg-sky-500/15 text-sky-300'}`}>{scopeName(a.scope)}</span>
149:                   <p className="mt-1 font-mono text-xs text-slate-600">{a.slug}</p>
150:                 </td>
151:                 <td className="px-3 py-3 font-semibold text-slate-100">{a.name}</td>
152:                 <td className="px-3 py-3 text-slate-300">{formatGrant(a)}</td>
153:                 <td className="px-3 py-3 text-center font-bold text-white">{brl(a.price_monthly)}</td>
154:                 <td className="px-3 py-3 text-center">
155:                   <span className={`rounded-full px-2 py-0.5 text-xs font-bold ${a.is_upsell ? 'bg-amber-500/15 text-amber-300' : 'bg-slate-500/15 text-slate-400'}`}>{a.is_upsell ? 'Sim' : 'Não'}</span>
156:                 </td>
157:                 <td className="px-3 py-3 text-center">
158:                   <span className={`rounded-full px-2 py-0.5 text-xs font-bold ${a.is_active ? 'bg-emerald-500/15 text-emerald-300' : 'bg-slate-500/15 text-slate-400'}`}>{a.is_active ? 'Ativo' : 'Inativo'}</span>
159:                 </td>
160:                 <td className="px-3 py-3">
161:                   <div className="flex justify-center gap-1">
162:                     <button type="button" title="Editar" onClick={() => openEdit(a)} className="cursor-pointer rounded-lg p-2 text-slate-300 hover:bg-white/10"><Pencil size={15} /></button>
163:                     <button type="button" title="Excluir" onClick={() => remove(a)} className="cursor-pointer rounded-lg p-2 text-rose-400 hover:bg-white/10"><Trash2 size={15} /></button>
164:                   </div>
165:                 </td>
166:               </tr>
167:             ))}
168:           </tbody>
169:         </table>
170:       </div>
171: 
172:       {form && (
173:         <div className="fixed inset-0 z-[70] flex items-center justify-center bg-black/70 p-4" onClick={() => !busy && setForm(null)}>
174:           <div className="max-h-[90vh] w-full max-w-xl space-y-4 overflow-y-auto rounded-2xl border border-slate-700 bg-[#0f1322] p-6" onClick={(e) => e.stopPropagation()}>
175:             <div className="flex items-center justify-between">
176:               <h3 className="text-lg font-black text-white">{editId ? 'Editar add-on' : 'Novo add-on'}</h3>
177:               <button type="button" aria-label="Fechar" onClick={() => setForm(null)} className="cursor-pointer text-slate-400 hover:text-white"><X size={18} /></button>
178:             </div>
179: 
180:             <label className="block"><span className={LBL}>Nome</span>
181:               <input value={form.name} onChange={(e) => set('name', e.target.value)} className={IN} placeholder="+10 GB de Armazenamento" autoFocus /></label>
182:             <label className="block"><span className={LBL}>Descrição (opcional)</span>
183:               <input value={form.description} onChange={(e) => set('description', e.target.value)} className={IN} /></label>
184: 
185:             <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
186:               <label className="block"><span className={LBL}>Escopo</span>
187:                 <select value={form.scope} onChange={(e) => set('scope', e.target.value)} className={`${IN} cursor-pointer`}>
188:                   {scopes.map((s) => <option key={s.slug} value={s.slug}>{s.name}</option>)}
189:                 </select></label>
190:               <label className="block"><span className={LBL}>Preço mensal (R$)</span>
191:                 <input value={form.price} onChange={(e) => set('price', e.target.value)} className={IN} placeholder="19,90" inputMode="decimal" /></label>
192:               <label className="block"><span className={LBL}>Recurso incrementado</span>
193:                 <select value={form.target_resource} onChange={(e) => set('target_resource', e.target.value as Resource)} className={`${IN} cursor-pointer`}>
194:                   {(Object.keys(RESOURCE_LABEL) as Resource[]).map((r) => <option key={r} value={r}>{RESOURCE_LABEL[r]}</option>)}
195:                 </select></label>
196:               <label className="block"><span className={LBL}>Quantidade concedida</span>
197:                 <input value={form.amount} onChange={(e) => set('amount', e.target.value)} className={IN} placeholder="10 ou 15000" inputMode="decimal" /></label>
198:             </div>
199:             {form.target_resource === 'custom' && <p className="text-[13px] text-slate-500">Recurso personalizado não altera limites automaticamente.</p>}
200: 
201:             <Switch on={form.is_active} onChange={(v) => set('is_active', v)} label={form.is_active ? 'Ativo' : 'Inativo'} />
202: 
203:             <div className="space-y-3 rounded-xl border border-slate-800 p-4">
204:               <label className="flex cursor-pointer items-center gap-2 text-[15px] font-semibold text-slate-100">
205:                 <input type="checkbox" checked={form.is_upsell} onChange={(e) => set('is_upsell', e.target.checked)} />
206:                 Oferecer como Upsell / Order Bump no Checkout
207:               </label>
208:               {form.is_upsell && (
209:                 <>
210:                   <label className="block"><span className={LBL}>Onde oferecer</span>
211:                     <select value={form.upsell_module_target} onChange={(e) => set('upsell_module_target', e.target.value)} className={`${IN} cursor-pointer`}>
212:                       {targets.map((t) => <option key={t.slug} value={t.slug}>{t.name}</option>)}
213:                     </select></label>
214:                   <label className="block"><span className={LBL}>Frase de destaque</span>
215:                     <input value={form.upsell_headline} onChange={(e) => set('upsell_headline', e.target.value)} className={IN}
216:                       placeholder="Oferta exclusiva: adicione 15.000 plays por apenas R$ 19,90/mês" /></label>
217:                 </>
218:               )}
219:             </div>
220: 
221:             <div className="flex justify-end gap-2 border-t border-slate-800 pt-4">
222:               <button type="button" disabled={busy} onClick={() => setForm(null)} className="cursor-pointer rounded-lg bg-white/5 px-4 py-2 text-[13px] font-semibold text-slate-200 hover:bg-white/10">Cancelar</button>
223:               <button type="button" disabled={busy} onClick={save} className="cursor-pointer rounded-lg bg-[#0094eb] px-5 py-2 text-[13px] font-bold text-white disabled:opacity-40">
224:                 {busy ? 'Salvando...' : 'Salvar add-on'}
225:               </button>
226:             </div>
227:           </div>
228:         </div>
229:       )}
230:       {dialog}
231:     </div>
232:   );
233: }
